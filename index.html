function buildDynamicZipListUI() { const box = document.getElementById('zipScrollBox'); if(!box) return; box.innerHTML = ""; const availableZips = Object.keys(database).sort(); if(availableZips.length === 0) return; const cfg = CITY_CONFIG[currentCityKey]; const hAll = document.createElement('label'); hAll.className = "zip-item"; hAll.innerHTML = ` ALL ZIPs`; box.appendChild(hAll); let hasMOU = availableZips.some(z => cfg.stationMOU.includes(z)); if (hasMOU && cfg.stationDCU.length > 0) { const hMOU = document.createElement('label'); hMOU.className = "zip-item"; hMOU.innerHTML = ` ${cfg.mouName.toUpperCase()}`; box.appendChild(hMOU); } let hasDCU = availableZips.some(z => cfg.stationDCU.includes(z)); if (hasDCU && cfg.stationDCU.length > 0) { const hDCU = document.createElement('label'); hDCU.className = "zip-item"; hDCU.innerHTML = ` ${cfg.dcuName.toUpperCase()}`; box.appendChild(hDCU); } availableZips.forEach(zip => { const lbl = document.createElement('label'); lbl.className = "zip-item"; // THIS IS THE EXACT INTERSECTION THAT MAKES IT READ THE NAMES: const displayLabel = (cfg.zipLabels && cfg.zipLabels[zip]) ? cfg.zipLabels[zip] : zip; lbl.innerHTML = ` ${displayLabel}`; box.appendChild(lbl); }); }
