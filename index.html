Initial configuration and data state // (This remains the same to ensure your picklist stays active) const addressData = [ { zip: "98663", street: "East 20th Street" }, { zip: "98663", street: "East 26th Street" }, { zip: "98663", street: "East 29th Street" }, { zip: "98663", street: "Northeast 2nd Avenue" }, { zip: "98661", street: "East 5th Street" } ]; /** * Updates the street picklist based on the selected Zip Code. * Includes the fix for the "E" vs "NE" and numeric sorting anomaly. */ function updateStreetPicklist(selectedZip) { const picklist = document.getElementById('streetPicklist'); // Safety check: Ensure the picklist element exists before proceeding if (!picklist) return; // STEP A: Filter by Zip Code (Preserves existing functionality) const filteredAddresses = addressData.filter(item => item.zip === selectedZip); // STEP B: THE FIX - Natural Numeric Sort // This compares the strings by looking for numbers first. // It treats "2" as smaller than "20", even with "Northeast" in front. filteredAddresses.sort((a, b) => { return a.street.localeCompare(b.street, undefined, { numeric: true, sensitivity: 'base' }); }); // STEP C: Clear and Rebuild the Picklist // This keeps the "Select a Street..." default and prevents duplication picklist.innerHTML = ''; filteredAddresses.forEach(item => { const option = document.createElement('option'); option.value = item.street; option.textContent = item.street; picklist.appendChild(option); }); } // 2. Event Listener Logic // Listens for the Zip Code change to trigger the new sorting logic document.getElementById('zipInput').addEventListener('change', (e) => { const zipValue = e.target.value; // Standard 5-digit validation before updating if (zipValue.length === 5) { updateStreetPicklist(zipValue); } }); // 3. Optional: Initial check if a zip is already typed in on load window.addEventListener('DOMContentLoaded', () => { const zipInput = document.getElementById('zipInput'); if (zipInput && zipInput.value.length === 5) { updateStreetPicklist(zipInput.value); } });
